/**
 * Displays the user's roster on the battle screen.
 */
#showRoster() {
  // Display for roster panel
  rectMode(CENTER);
  fill('#8A946F');
  stroke('black');
  strokeWeight(2);
  let rosterSpace = 32;
  for (let x = 0; x < 6; x++) {
    rect(rosterSpace, height - 32, 60, 60);
    rosterSpace += 60;
  }

  // display roster sprites
  let spacing = 60;
  let x = 32;
  let y = 568;
  for (let j = 0; j < playerRoster.maxSize; j++) {
    image(playerRoster.next().sprite_front, x, y);
    x += spacing;
  }

  // selection square is yellow until a selection is made
  let r, g, b;
  if (batEng.turn && batEng.switchAllowed) {

    // If the current roster rotation is on the Player's current monster
    // make the selection box green
    if (playerRoster.peek() == batEng.currentMonsterPlayer) {
      r = 0;
      g = 255;
      b = 0;
    }

    // Otherwise, make selection box yellow
    else {
      r = 255;
      g = 251;
      b = 0;
    }

    if (playerRoster.peek().current_hp == 0) { // Selecton box is red if monster is unavailable
      r = 255;
      g = 0;
      b = 0;
    }
  }
  fill(r, g, b, 80)
  rect(32, height - 32, 60, 60);
}

DNS Client and Server

A DNS client in C that follows RFC 1035. Messages are encoded and parsed by hand, including name compression, for A, AAAA, MX, NS, CNAME, and SOA records. The server side answers lookups from a local binary record database.

View code 
int
decode_name (const uint8_t *buf, int offset, char *out)
{
  int out_pos = 0;
  int return_offset = -1; // set the first time we follow a compression pointer
  int jumps = 0;

  while (1)
    {
      uint8_t len = buf[offset];

      if (len == 0)
        {
          // if we never jumped, resume just past this null byte
          if (return_offset == -1)
            return_offset = offset + 1;
          break;
        }

      if ((len & 0xC0) == 0xC0)
        {
          // Compression pointer: top 2 bits are set.
          // The offset to jump to is the lower 6 bits of this byte
          // combined with all 8 bits of the next byte.
          if (return_offset == -1)
            return_offset = offset + 2; // caller resumes past these 2 bytes
          offset = ntohs (*(uint16_t *)&buf[offset]) & 0x3FFF;
          if (++jumps > 10)
            break; // guard against malformed looping pointers
          continue;
        }

      // Regular label: len bytes of name text follow
      offset++;
      if (out_pos > 0)
        out[out_pos++] = '.';
      memcpy (out + out_pos, buf + offset, len);
      out_pos += len;
      offset += len;
    }

  // append a trailing dot to decoded names
  out[out_pos++] = '.';
  out[out_pos] = '\0';

  return return_offset;
}
